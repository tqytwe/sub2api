### 步骤 3：添加 computed 属性

在 `<script setup>` 中添加：

```typescript
const couponPercentage = computed(() => {
  if (!status.value?.coupon_weight_bp) return 0
  return Math.round((status.value.coupon_weight_bp / 10000) * 100)
})

const redeemPercentage = computed(() => {
  if (!status.value?.redeem_code_weight_bp) return 0
  return Math.round((status.value.redeem_code_weight_bp / 10000) * 100)
})

const balancePercentage = computed(() => {
  if (!status.value?.balance_weight_bp) return 0
  return Math.round((status.value.balance_weight_bp / 10000) * 100)
})
```
